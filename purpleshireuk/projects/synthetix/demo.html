main.py
from synthetix import Swarm, Agent

# Initialize the swarm
swarm = Swarm(api_key="sk_live_...")

# Define a specialized agent
researcher = Agent(
    role="Senior Researcher",
    goal="Analyze market trends for crypto",
    tools=["web_scrape", "sentiment_analysis"],
    verbose=True
)

# Execute task
task = """
Find the top 3 emerging DeFi protocols
and summarize their whitepapers.
"""

report = swarm.execute(agents=[researcher], task=task)
print(report)
